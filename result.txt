package com.example.aas.block;

import com.example.aas.AASMod;
import net.minecraft.world.level.block.Block;
import net.minecraft.world.level.block.entity.BlockEntityType;
import net.minecraftforge.eventbus.api.IEventBus;
import net.minecraftforge.registries.DeferredRegister;
import net.minecraftforge.registries.ForgeRegistries;
import net.minecraftforge.registries.RegistryObject;

public class ModBlocks {
    public static final DeferredRegister<Block> BLOCKS = DeferredRegister.create(ForgeRegistries.BLOCKS, "aas");
    public static final DeferredRegister<BlockEntityType<?>> BLOCK_ENTITIES = DeferredRegister.create(ForgeRegistries.BLOCK_ENTITY_TYPES, "aas");

    public static final RegistryObject<Block> BLUE_RALLY_BLOCK = BLOCKS.register("blue_rally", RallyPointBlock::new);
    public static final RegistryObject<Block> RED_RALLY_BLOCK = BLOCKS.register("red_rally", RallyPointBlock::new);

    public static final RegistryObject<BlockEntityType<RallyPointBlockEntity>> RALLY_BE = BLOCK_ENTITIES.register("rally_be",
            () -> BlockEntityType.Builder.of(RallyPointBlockEntity::new, BLUE_RALLY_BLOCK.get(), RED_RALLY_BLOCK.get()).build(null));

    public static void register(IEventBus bus) {
        BLOCKS.register(bus);
        BLOCK_ENTITIES.register(bus);
    }
}

package com.example.aas.block;

import com.example.aas.world.AASWorldData;
import net.minecraft.core.BlockPos;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.world.level.block.entity.BlockEntity;
import net.minecraft.world.level.block.state.BlockState;

public class RallyPointBlockEntity extends BlockEntity {
    private int lifeTime = 0;
    private final int MAX_LIFE = 5 * 60 * 20; // 5 РјРёРЅСѓС‚ РІ С‚РёРєР°С…

    public RallyPointBlockEntity(BlockPos pos, BlockState state) {
        super(ModBlocks.RALLY_BE.get(), pos, state);
    }

    public void tick(ServerLevel level) {
        lifeTime++;
        if (lifeTime >= MAX_LIFE) {
            // Р’СЂРµРјСЏ РІС‹С€Р»Рѕ - СѓРґР°Р»СЏРµРј Р±Р»РѕРє
            level.removeBlock(worldPosition, false);

            // РЈРґР°Р»СЏРµРј Р·Р°РїРёСЃСЊ РёР· РґР°РЅРЅС‹С… РјРёСЂР°
            AASWorldData data = AASWorldData.get(level);
            if (data.blueRallyPos != null && data.blueRallyPos.equals(worldPosition)) {
                data.blueRallyPos = null;
                data.setDirty();
            } else if (data.redRallyPos != null && data.redRallyPos.equals(worldPosition)) {
                data.redRallyPos = null;
                data.setDirty();
            }
        }
    }
}

package com.example.aas.client;

import com.example.aas.network.PacketHandler;
import com.example.aas.network.PacketRespawnRequest;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.client.gui.components.Button;
import net.minecraft.client.gui.screens.DeathScreen;
import net.minecraft.network.chat.Component;

public class AASDeathScreen extends DeathScreen {
    private int respawnTimeTotal = 10; // РЎС‚Р°РЅРґР°СЂС‚РЅРѕРµ Р·РЅР°С‡РµРЅРёРµ
    private long deathTimestamp;

    public AASDeathScreen(Component cause, boolean hardcore) {
        super(cause, hardcore);
        this.deathTimestamp = System.currentTimeMillis();
        // Р‘РµСЂРµРј С‚Р°Р№РјРµСЂ, РєРѕС‚РѕСЂС‹Р№ РїСЂРёС€РµР» СЃ СЃРµСЂРІРµСЂР°
        this.respawnTimeTotal = ClientData.RESPAWN_TIME;
    }

    @Override
    public boolean isPauseScreen() { return false; }

    @Override
    protected void init() {
        super.init();
        this.clearWidgets();

        int centerX = this.width / 2;
        int centerY = this.height / 2;

        // РљРЅРѕРїРєР° MAIN SPAWN (РўР•РџР•Р Р¬ РџРћ Р¦Р•РќРўР РЈ)
        // РЁРёСЂРёРЅР° РєРЅРѕРїРєРё 100, Р·РЅР°С‡РёС‚ СЃРјРµС‰РµРЅРёРµ РІР»РµРІРѕ РґРѕР»Р¶РЅРѕ Р±С‹С‚СЊ 50
        this.addRenderableWidget(Button.builder(Component.literal("MAIN SPAWN"), button -> {
            PacketHandler.INSTANCE.sendToServer(new PacketRespawnRequest("MAIN"));
            this.minecraft.player.respawn();
            this.minecraft.setScreen(null);
        }).bounds(centerX - 50, centerY + 30, 100, 20).build());

        // РљРЅРѕРїРєСѓ "RALLY POINT" РњР« РЈР”РђР›РР›Р, РєР°Рє С‚С‹ РїСЂРѕСЃРёР».

        this.addRenderableWidget(Button.builder(Component.literal("Disconnect"), button -> {
            this.minecraft.level.disconnect();
            this.minecraft.clearLevel();
            this.minecraft.setScreen(new net.minecraft.client.gui.screens.TitleScreen());
        }).bounds(centerX - 50, centerY + 55, 100, 20).build());
    }

    @Override
    public void render(GuiGraphics gui, int mouseX, int mouseY, float partialTick) {
        super.render(gui, mouseX, mouseY, partialTick);
        long secondsLeft = respawnTimeTotal - ((System.currentTimeMillis() - deathTimestamp) / 1000);
        if (secondsLeft < 0) secondsLeft = 0;

        gui.drawCenteredString(this.font, "RESPAWN IN: " + secondsLeft, this.width / 2, this.height / 2 + 10, 0xFFAA00);

        boolean canRespawn = (secondsLeft == 0);
        this.children().forEach(c -> {
            if (c instanceof Button b) {
                if (!b.getMessage().getString().equals("Disconnect")) {
                    b.active = canRespawn;
                }
            }
        });
    }
}

import net.minecraft.client.Minecraft;

@Mod.EventBusSubscriber(modid = "aas", value = Dist.CLIENT, bus = Mod.EventBusSubscriber.Bus.FORGE)
public class AASOverlay {

    @SubscribeEvent
    public static void onRenderOverlay(RenderGuiOverlayEvent.Post event) {
        if (event.getOverlay() != VanillaGuiOverlay.HOTBAR.type()) return;

        GuiGraphics gui = event.getGuiGraphics();
        Minecraft mc = Minecraft.getInstance();
        int width = mc.getWindow().getGuiScaledWidth();
        int height = mc.getWindow().getGuiScaledHeight();

        // 1. Р РёСЃСѓРµРј С‚РёРєРµС‚С‹ СЃРІРµСЂС…Сѓ
        renderTickets(gui, mc, width);

        if (ClientData.isInsidePoint) {
            renderCapturePoint(gui, mc, width, height);
        }
    }

    private static void renderTickets(GuiGraphics gui, Minecraft mc, int width) {
        int boxWidth = 45; int boxHeight = 14; int gap = 4; int topOffset = 5;
        int centerX = width / 2;

        // Р›РѕРіРёРєР° РјРёРіР°РЅРёСЏ (РєР°Р¶РґС‹Рµ 0.5 СЃРµРєСѓРЅРґС‹ РјРµРЅСЏРµС‚СЃСЏ СЃРѕСЃС‚РѕСЏРЅРёРµ)
        boolean blinkOn = (System.currentTimeMillis() / 500) % 2 == 0;

        // Р¦РІРµС‚Р° С‚РµРєСЃС‚Р°
        int normalWhite = 0xFFFFFFFF;
        int alarmRed = 0xFFFF5555; // РЎРІРµС‚Р»Рѕ-РєСЂР°СЃРЅС‹Р№ РґР»СЏ С‚СЂРµРІРѕРіРё

        // --- РЎРРќРР• ---
        int blueX = centerX - boxWidth - (gap / 2);
        gui.fill(blueX - 1, topOffset - 1, blueX + boxWidth + 1, topOffset + boxHeight + 1, 0xFF000000);
        gui.fill(blueX, topOffset, blueX + boxWidth, topOffset + boxHeight, 0xCC3366CC);

        String blueText = String.valueOf(ClientData.BLUE_TICKETS);
        int blueTextColor = normalWhite;

        // Р•СЃР»Рё РЎРёРЅРёРµ "РєСЂРѕРІРѕС‚РѕС‡Р°С‚", РјРёРіР°РµРј
        if (ClientData.blueBleeding) {
            blueTextColor = blinkOn ? alarmRed : normalWhite;
        }

        gui.drawString(mc.font, blueText, blueX + (boxWidth - mc.font.width(blueText)) / 2, topOffset + 3, blueTextColor, false);


        // --- РљР РђРЎРќР«Р• ---
        int redX = centerX + (gap / 2);
        gui.fill(redX - 1, topOffset - 1, redX + boxWidth + 1, topOffset + boxHeight + 1, 0xFF000000);
        gui.fill(redX, topOffset, redX + boxWidth, topOffset + boxHeight, 0xCCCC3333);

        String redText = String.valueOf(ClientData.RED_TICKETS);
        int redTextColor = normalWhite;

        // Р•СЃР»Рё РљСЂР°СЃРЅС‹Рµ "РєСЂРѕРІРѕС‚РѕС‡Р°С‚", РјРёРіР°РµРј
        if (ClientData.redBleeding) {
            redTextColor = blinkOn ? alarmRed : normalWhite;
        }

        gui.drawString(mc.font, redText, redX + (boxWidth - mc.font.width(redText)) / 2, topOffset + 3, redTextColor, false);
    }

    private static void renderCapturePoint(GuiGraphics gui, Minecraft mc, int width, int height) {
        int centerX = width / 2;
        int bottomY = height - 55;

        // === Р¤Р›РђР“-РџР РЇРњРћРЈР“РћР›Р¬РќРРљ ===
        int flagWidth = 32;
        int flagHeight = 18;
        int flagX = centerX - (flagWidth / 2);
        int flagY = bottomY - flagHeight;

        int color = 0xFFFFFFFF; // NEUTRAL
        if (ClientData.pointOwner.equals("BLUE")) color = 0xFF3366CC;
        if (ClientData.pointOwner.equals("RED")) color = 0xFFCC3333;

        // Р Р°РјРєР° Рё С„РѕРЅ С„Р»Р°РіР°
        gui.fill(flagX - 1, flagY - 1, flagX + flagWidth + 1, flagY + flagHeight + 1, 0xFF000000);

        // РќР°Р·РІР°РЅРёРµ С‚РѕС‡РєРё
        String name = ClientData.pointName;
        int nameW = mc.font.width(name);
        gui.drawString(mc.font, name, centerX - (nameW / 2), flagY - 10, 0xFFFFFFFF, true);

        int barsTotalWidth = 80;
        int barHeight = 4;
        int barsStartX = centerX - (barsTotalWidth / 2);
        int barsY = flagY + flagHeight + 3;

        // === Р›РћР“РРљРђ РћРўРћР‘Р РђР–Р•РќРРЇ РЎРўРђРўРЈРЎРђ ===

        if (ClientData.isLocked) {
            // 1. Р•СЃР»Рё С‚РѕС‡РєР° Р·Р°Р±Р»РѕРєРёСЂРѕРІР°РЅР° РїРѕ РїСЂРёРѕСЂРёС‚РµС‚Сѓ
            String lockedText = "LOCKED!";
            String subText = "Capture " + ClientData.nextObjectiveName + " first";

            gui.drawString(mc.font, lockedText, centerX - (mc.font.width(lockedText) / 2), barsY, 0xFFFF5555, true);
            gui.drawString(mc.font, subText, centerX - (mc.font.width(subText) / 2), barsY + 10, 0xFFAAAAAA, true);

        } else if (ClientData.isContested) {
            // 2. Р•СЃР»Рё РЅР° С‚РѕС‡РєРµ РІСЂР°РіРё (Р±РѕР№)

            String subText = "Enemies nearby";

            gui.drawString(mc.font, contestText, centerX - (mc.font.width(contestText) / 2), barsY, 0xFFFFAA00, true);
            gui.drawString(mc.font, subText, centerX - (mc.font.width(subText) / 2), barsY + 10, 0xFFFFFF55, true);

        } else {
            // 3. РРЅР°С‡Рµ СЂРёСЃСѓРµРј РїРѕР»РѕСЃРєРё Р·Р°С…РІР°С‚Р°
            int barGap = 2;
            int singleBarWidth = (barsTotalWidth - (barGap * 3)) / 4;

                int currentBarX = barsStartX + (i * (singleBarWidth + barGap));
                // Р¤РѕРЅ РїРѕР»РѕСЃРєРё

                    gui.fill(currentBarX, barsY, currentBarX + fillWidth, barsY + barHeight, barColor);
                }
            }
        }
    }
}

    @SubscribeEvent
    public static void onMouseInput(InputEvent.MouseButton event) {
        Minecraft mc = Minecraft.getInstance();
        if (mc.player == null || mc.level == null) return;
        if (event.getButton() == GLFW.GLFW_MOUSE_BUTTON_MIDDLE && event.getAction() == GLFW.GLFW_PRESS) {
            if (mc.player.getMainHandItem().getItem() instanceof VehicleMarkerItem) {
                HitResult result = mc.hitResult;
                if (result != null && result.getType() == HitResult.Type.ENTITY) {
                    EntityHitResult entityResult = (EntityHitResult) result;
                    Entity target = entityResult.getEntity();
                    PacketHandler.INSTANCE.sendToServer(new PacketApplyMarker(target.getId()));
                }
            }
        }
    }

    // === РџРћР”РњР•РќРђ Р­РљР РђРќРђ РЎРњР•Р РўР ===
    @SubscribeEvent
    public static void onOpenGui(ScreenEvent.Opening event) {
        if (event.getScreen() instanceof DeathScreen && !(event.getScreen() instanceof AASDeathScreen)) {
            Component cause = null;
            if (Minecraft.getInstance().player != null) {
                cause = Minecraft.getInstance().player.getCombatTracker().getDeathMessage();
            }
            // Р—Р°РјРµРЅСЏРµРј СЌРєСЂР°РЅ РЅР° РЅР°С€
            event.setNewScreen(new AASDeathScreen(cause, false));
        }
    }

    @SubscribeEvent
    public static void onKeyInput(InputEvent.Key event) {
        Minecraft mc = Minecraft.getInstance();
        if (mc.player == null) return;

        if (event.getAction() == GLFW.GLFW_PRESS) {

    private static int setRespawnTime(CommandSourceStack source, int seconds) {

    private static int joinTeam(CommandSourceStack source, String teamName, ServerPlayer player) {
        Scoreboard scoreboard = source.getServer().getScoreboard();
        String internalTeamName = teamName.equalsIgnoreCase("blue") ? "Blue" : "Red";
        ChatFormatting color = teamName.equalsIgnoreCase("blue") ? ChatFormatting.BLUE : ChatFormatting.RED;
        PlayerTeam team = scoreboard.getPlayerTeam(internalTeamName);
        if (team == null) team = scoreboard.addPlayerTeam(internalTeamName);
        team.setColor(color);
        team.setAllowFriendlyFire(false);
        scoreboard.addPlayerToTeam(player.getScoreboardName(), team);
        source.sendSuccess(() -> Component.literal("Player joined " + internalTeamName).withStyle(color), true);
        return 1;
    }

    private static int addPoint(CommandSourceStack source, BlockPos pos1, BlockPos pos2, String name, int bp, int rp, int time, int penalty) {
        AASWorldData data = AASWorldData.get(source.getServer().overworld());
        double minX = Math.min(pos1.getX(), pos2.getX()); double minY = Math.min(pos1.getY(), pos2.getY()); double minZ = Math.min(pos1.getZ(), pos2.getZ());
        double maxX = Math.max(pos1.getX(), pos2.getX()) + 1; double maxY = Math.max(pos1.getY(), pos2.getY()) + 1; double maxZ = Math.max(pos1.getZ(), pos2.getZ()) + 1;
        AABB area = new AABB(minX, minY, minZ, maxX, maxY, maxZ);
        data.capturePoints.add(new AASWorldData.CapturePoint(name, area, bp, rp, time, penalty));
        data.setDirty();
        source.sendSuccess(() -> Component.literal("Point '" + name + "' added!").withStyle(ChatFormatting.GREEN), true);
        return 1;
    }

    private static int removePoint(CommandSourceStack source, String name) {
        AASWorldData data = AASWorldData.get(source.getServer().overworld());
        boolean removed = data.capturePoints.removeIf(p -> p.name.equals(name));
        if (removed) { data.setDirty(); source.sendSuccess(() -> Component.literal("Point '" + name + "' removed!").withStyle(ChatFormatting.RED), true); return 1; }
        else { source.sendFailure(Component.literal("Point '" + name + "' not found!")); return 0; }
    }

    private static int setTeamSpawn(CommandSourceStack source, String teamName, BlockPos pos) {
        AASWorldData data = AASWorldData.get(source.getServer().overworld());

        data.setDirty();
        return 1;
    }

    @SubscribeEvent
    public static void onPlayerClone(PlayerEvent.Clone event) {
        if (event.isWasDeath()) {
            ServerPlayer original = (ServerPlayer) event.getOriginal();
            ServerPlayer newPlayer = (ServerPlayer) event.getEntity();
            String teamName = null;
            if (original.getTeam() != null) teamName = original.getTeam().getName();
            if (teamName == null) teamName = pendingTeams.get(newPlayer.getUUID());
            if (teamName != null) {
                Scoreboard scoreboard = newPlayer.getScoreboard();
                PlayerTeam team = scoreboard.getPlayerTeam(teamName);
                if (team != null) scoreboard.addPlayerToTeam(newPlayer.getScoreboardName(), team);
            }
        }
    }

    @SubscribeEvent
    public static void onPlayerRespawn(PlayerEvent.PlayerRespawnEvent event) {
        if (event.getEntity().level().isClientSide) return;
        ServerPlayer newPlayer = (ServerPlayer) event.getEntity();
        if (newPlayer.gameMode.getGameModeForPlayer() != GameType.CREATIVE) newPlayer.setGameMode(GameType.SURVIVAL);
        newPlayer.connection.send(new ClientboundPlayerAbilitiesPacket(newPlayer.getAbilities()));
        pendingRespawnLocations.remove(newPlayer.getUUID());
        pendingTeams.remove(newPlayer.getUUID());
    }

    private static void processEntityLoss(Entity entity) {
        // Р—Р°С‰РёС‚Р°: РёРЅРѕРіРґР° РїСЂРё РІС‹С…РѕРґРµ РёР· РјРёСЂР° level РјРѕР¶РµС‚ Р±С‹С‚СЊ null

        AASWorldData data = AASWorldData.get(overworld);

        if (data.blueTickets <= 0 || data.redTickets <= 0) return;

        boolean ticketsChanged = false;

        // Рђ) РР“Р РћРљ
        if (entity instanceof ServerPlayer player) {
            if (player.getTeam() != null) {
                String teamName = player.getTeam().getName();
                if (teamName.equalsIgnoreCase("Blue")) {
                    data.blueTickets -= 2; ticketsChanged = true;
                } else if (teamName.equalsIgnoreCase("Red")) {
                    data.redTickets -= 2; ticketsChanged = true;
                }
            }
        }

        // Р‘) РўР•РҐРќРРљРђ (Р›Р®Р‘РђРЇ РЎРЈР©РќРћРЎРўР¬ РЎ РњР•РўРљРћР™)
        // РџСЂРѕРІРµСЂСЏРµРј РЅР°Р»РёС‡РёРµ РјРµС‚РєРё С†РµРЅС‹ (РєРѕС‚РѕСЂСѓСЋ РјС‹ СЃС‚Р°РІРёРј РїСЂРµРґРјРµС‚РѕРј)
        if (entity.getPersistentData().contains("AAS_TicketPenalty")) {
            int penalty = entity.getPersistentData().getInt("AAS_TicketPenalty");
            String vTeam = entity.getPersistentData().getString("AAS_VehicleTeam");
            String vType = entity.getPersistentData().getString("AAS_VehicleType");

            // РЈРґР°Р»СЏРµРј С‚РµРі, С‡С‚РѕР±С‹ РЅРµ Р·Р°СЃС‡РёС‚Р°Р»Рѕ РґРІР°Р¶РґС‹
            entity.getPersistentData().remove("AAS_TicketPenalty");

            if (penalty > 0) {
                if (vTeam.equalsIgnoreCase("BLUE")) {
                    data.blueTickets -= penalty;
                    ticketsChanged = true;
                    broadcastMessage(overworld, "BLUE lost " + vType + " (-" + penalty + ")", ChatFormatting.RED);
                }
                else if (vTeam.equalsIgnoreCase("RED")) {
                    data.redTickets -= penalty;
                    ticketsChanged = true;
                    broadcastMessage(overworld, "RED lost " + vType + " (-" + penalty + ")", ChatFormatting.BLUE);
                }
            }
        }

        if (ticketsChanged) {
            checkGameOver(overworld, data);
            data.setDirty();
            sendSyncPacket(data);
        }
    }

    @SubscribeEvent
    public static void onWorldTick(TickEvent.LevelTickEvent event) {
        if (event.level.isClientSide || event.phase != TickEvent.Phase.END) return;
        if (event.level.dimension() != Level.OVERWORLD) return;

        ServerLevel level = (ServerLevel) event.level;
        AASWorldData data = AASWorldData.get(level);
        boolean gameEnded = (data.blueTickets <= 0 || data.redTickets <= 0);

        boolean blueIsBleeding = false;
        boolean redIsBleeding = false;

        if (!gameEnded && !data.capturePoints.isEmpty()) {
            long totalPoints = data.capturePoints.size();
            long blueOwned = data.capturePoints.stream().filter(p -> p.owner.equals("BLUE")).count();
            long redOwned = data.capturePoints.stream().filter(p -> p.owner.equals("RED")).count();

            if (blueOwned == totalPoints) redIsBleeding = true;
            else if (redOwned == totalPoints) blueIsBleeding = true;

            if (level.getGameTime() % 40 == 0) {
                if (redIsBleeding) data.redTickets -= 1;
                if (blueIsBleeding) data.blueTickets -= 1;
                if (redIsBleeding || blueIsBleeding) {
                    checkGameOver(level, data);
                    data.setDirty();
                }
            }
        }

        if (level.getGameTime() % 20 == 0) {
            sendSyncPacket(data, blueIsBleeding, redIsBleeding);
        }

        Set<UUID> playersInZones = new HashSet<>();

        for (AASWorldData.CapturePoint point : data.capturePoints) {
            List<ServerPlayer> players = level.getEntitiesOfClass(ServerPlayer.class, point.area);
            int blueCount = 0; int redCount = 0;
            for (ServerPlayer p : players) {
                if (!p.isAlive() || p.isSpectator()) continue;
                playersInZones.add(p.getUUID());
                if (p.getTeam() != null) {
                    String tName = p.getTeam().getName();
                    if (tName.equalsIgnoreCase("Blue")) blueCount++; else if (tName.equalsIgnoreCase("Red")) redCount++;
                }
            }
            boolean contested = (blueCount > 0 && redCount > 0);
            for (ServerPlayer p : players) {
                if (!p.isAlive() || p.isSpectator()) continue;
                boolean locked = false; String requiredPointName = "";
                if (p.getTeam() != null) {
                    String team = p.getTeam().getName();
                    if (team.equalsIgnoreCase("Blue") && !canCapture(point, "BLUE", data)) { locked = true; requiredPointName = getPreviousPointName(point.bluePriority, "BLUE", data); }
                    else if (team.equalsIgnoreCase("Red") && !canCapture(point, "RED", data)) { locked = true; requiredPointName = getPreviousPointName(point.redPriority, "RED", data); }
                }
                PacketHandler.INSTANCE.send(PacketDistributor.PLAYER.with(() -> p), new PacketSyncPoint(true, point.name, point.owner, point.progress, locked, requiredPointName, contested));
            }

            if (gameEnded) continue;

            float captureSpeed = 1.0f / (point.captureTimeMinutes * 60 * 20);
            int dominance = Math.abs(blueCount - redCount);
            if (dominance > 0) {
                if (blueCount > redCount) { if (canCapture(point, "BLUE", data)) handleTeamInfluence(point, "BLUE", dominance, captureSpeed, data, level); }
                else { if (canCapture(point, "RED", data)) handleTeamInfluence(point, "RED", dominance, captureSpeed, data, level); }
            } else {
                if (blueCount == 0 && redCount == 0 && point.owner.equals("NEUTRAL") && point.progress > 0) {
                    point.progress -= (captureSpeed / 2.0f); if (point.progress < 0) point.progress = 0;
                }
            }
        }
        for (ServerPlayer player : level.players()) {
            if (!playersInZones.contains(player.getUUID())) {
                PacketHandler.INSTANCE.send(PacketDistributor.PLAYER.with(() -> player), new PacketSyncPoint(false, "", "", 0, false, "", false));
            }
        }
    }

    private static void checkGameOver(ServerLevel level, AASWorldData data) {
        if (data.blueTickets <= 0) {
            data.blueTickets = 0;
            sendWinTitle(level, "RED TEAM WINS!", ChatFormatting.RED);
        } else if (data.redTickets <= 0) {
            data.redTickets = 0;
            sendWinTitle(level, "BLUE TEAM WINS!", ChatFormatting.BLUE);
        }
    }

    private static void sendWinTitle(ServerLevel level, String text, ChatFormatting color) {
        Component title = Component.literal(text).withStyle(color).withStyle(ChatFormatting.BOLD);
        Component subtitle = Component.literal("Tickets depleted").withStyle(ChatFormatting.WHITE);
        for (ServerPlayer player : level.players()) {
            player.connection.send(new ClientboundSetTitlesAnimationPacket(10, 100, 20));
            player.connection.send(new ClientboundSetTitleTextPacket(title));

        }
    }

    private static void broadcastMessage(ServerLevel level, String text, ChatFormatting color) {

    private static void sendSyncPacket(AASWorldData data) {
        boolean blueIsBleeding = false; boolean redIsBleeding = false;
        if (!data.capturePoints.isEmpty()) {
            long total = data.capturePoints.size();
            long blueOwned = data.capturePoints.stream().filter(p -> p.owner.equals("BLUE")).count();
            long redOwned = data.capturePoints.stream().filter(p -> p.owner.equals("RED")).count();
            if (blueOwned == total) redIsBleeding = true;
            else if (redOwned == total) blueIsBleeding = true;
        }
        sendSyncPacket(data, blueIsBleeding, redIsBleeding);
    }

    private static void sendSyncPacket(AASWorldData data, boolean blueBleed, boolean redBleed) {
        boolean hasBlue = (data.blueRallyPos != null);
        boolean hasRed = (data.redRallyPos != null);
        PacketHandler.sendToAllClients(data.blueTickets, data.redTickets, hasBlue, hasRed, blueBleed, redBleed, data.respawnTimer);
    }

    private static void handleTeamInfluence(AASWorldData.CapturePoint point, String attackingTeam, int dominance, float speed, AASWorldData data, ServerLevel level) {
        float speedBoosted = speed * dominance;
        if (point.owner.equals("NEUTRAL")) {
            point.progress += speedBoosted;
            if (point.progress >= 1.0f) { point.progress = 1.0f; point.owner = attackingTeam; sendTitle(level, attackingTeam, point.name); }
        } else if (point.owner.equals(attackingTeam)) {
            if (point.progress < 1.0f) { point.progress += speedBoosted; if (point.progress > 1.0f) point.progress = 1.0f; }
        } else {
            point.progress -= speedBoosted;
            if (point.progress <= 0.0f) {
                if (point.owner.equals("BLUE")) data.blueTickets -= point.ticketPenalty; else data.redTickets -= point.ticketPenalty;
                checkGameOver(level, data); sendSyncPacket(data); broadcastMessage(level, point.name + " neutralized!", ChatFormatting.GRAY);
                point.owner = "NEUTRAL"; point.progress = 0.0f;
            }
        }
    }

    private static void sendTitle(ServerLevel level, String team, String pointName) {
        ChatFormatting color = team.equals("BLUE") ? ChatFormatting.BLUE : ChatFormatting.RED;
        Component title = Component.literal(team + " CAPTURED").withStyle(color).withStyle(ChatFormatting.BOLD);
        for (ServerPlayer player : level.players()) {
            player.connection.send(new ClientboundSetTitlesAnimationPacket(10, 60, 20));
            player.connection.send(new ClientboundSetTitleTextPacket(title));
            player.connection.send(new ClientboundSetSubtitleTextPacket(Component.literal(pointName)));
        }
    }

    private static boolean canCapture(AASWorldData.CapturePoint target, String team, AASWorldData data) {
        int currentPriority = team.equals("BLUE") ? target.bluePriority : target.redPriority;
        if (currentPriority <= 1) return true;
        int requiredPriority = currentPriority - 1;
        for (AASWorldData.CapturePoint p : data.capturePoints) {
            int pPriority = team.equals("BLUE") ? p.bluePriority : p.redPriority;
            if (pPriority == requiredPriority && p.owner.equals(team)) return true;

            ServerPlayer player = ctx.get().getSender();
            if (player == null) return;

            if (player.isAlive() && player.getHealth() > 0) return;

                if (msg.type.equals("MAIN")) {
                    targetPos = team.equalsIgnoreCase("Blue") ? data.blueSpawn : data.redSpawn;
                }
                else if (msg.type.equals("RALLY")) {

                        // РўСѓС‚ РёСЃРїРѕР»СЊР·СѓРµРј СЂРµР°Р»СЊРЅС‹Р№ СѓСЂРѕРІРµРЅСЊ РёРіСЂРѕРєР° РґР»СЏ РїСЂРѕРІРµСЂРєРё РІСЂР°РіРѕРІ (С‚.Рє. РІСЂР°РіРё С‚Р°Рј, РіРґРµ РјС‹)
                        AABB safetyBox = new AABB(rally).inflate(150);

                        if (!enemiesNearby) targetPos = rally;

                GameLogicEvents.pendingTeams.put(player.getUUID(), player.getTeam().getName());
            }
        });

    }

                buf.readBoolean(),
                buf.readUtf(),
                buf.readUtf(),
                buf.readFloat(),
                buf.readBoolean(),
                buf.readUtf(),
                buf.readBoolean()
        );
    }

        ctx.get().enqueueWork(() -> {
            // РћР±РЅРѕРІР»СЏРµРј РґР°РЅРЅС‹Рµ РЅР° РєР»РёРµРЅС‚Рµ РґР»СЏ РѕС‚СЂРёСЃРѕРІРєРё РѕРІРµСЂР»РµСЏ

            ClientData.pointOwner = msg.owner;
            ClientData.pointProgress = msg.progress;
            ClientData.isLocked = msg.isLocked;
            ClientData.nextObjectiveName = msg.nextObjective;
            ClientData.isContested = msg.isContested;
        });

package com.example.aas.world;

import net.minecraft.core.BlockPos;

import net.minecraft.nbt.Tag;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.world.level.saveddata.SavedData;
import net.minecraft.world.phys.AABB;

import java.util.ArrayList;
import java.util.List;

public class AASWorldData extends SavedData {
    public List<CapturePoint> capturePoints = new ArrayList<>();
    public BlockPos blueSpawn = null;
    public BlockPos redSpawn = null;
    public BlockPos blueRallyPos = null;
    public BlockPos redRallyPos = null;

    public int blueTickets = 800;
    public int redTickets = 800;

    // РќРћР’РћР•: РўР°Р№РјРµСЂ СЂРµСЃРїР°РІРЅР° (РїРѕ СѓРјРѕР»С‡Р°РЅРёСЋ 10 СЃРµРєСѓРЅРґ)
    public int respawnTimer = 10;

    @Override
    public CompoundTag save(CompoundTag tag) {
        tag.putInt("BlueTickets", blueTickets);
        tag.putInt("RedTickets", redTickets);
        tag.putInt("RespawnTimer", respawnTimer); // РЎРѕС…СЂР°РЅСЏРµРј

        if (blueRallyPos != null) tag.putLong("BlueRally", blueRallyPos.asLong());
        if (redRallyPos != null) tag.putLong("RedRally", redRallyPos.asLong());
        if (blueSpawn != null) tag.putLong("BlueSpawn", blueSpawn.asLong());
        if (redSpawn != null) tag.putLong("RedSpawn", redSpawn.asLong());

        ListTag list = new ListTag();
        for (CapturePoint point : capturePoints) {
            list.add(point.save());
        }
        tag.put("CapturePoints", list);

        return tag;
    }

    public static AASWorldData load(CompoundTag tag) {
        AASWorldData data = new AASWorldData();

        if (tag.contains("BlueTickets")) data.blueTickets = tag.getInt("BlueTickets");
        if (tag.contains("RedTickets")) data.redTickets = tag.getInt("RedTickets");
        // Р—Р°РіСЂСѓР¶Р°РµРј
        if (tag.contains("RespawnTimer")) data.respawnTimer = tag.getInt("RespawnTimer");

        if (tag.contains("BlueRally")) data.blueRallyPos = BlockPos.of(tag.getLong("BlueRally"));
        if (tag.contains("RedRally")) data.redRallyPos = BlockPos.of(tag.getLong("RedRally"));
        if (tag.contains("BlueSpawn")) data.blueSpawn = BlockPos.of(tag.getLong("BlueSpawn"));
        if (tag.contains("RedSpawn")) data.redSpawn = BlockPos.of(tag.getLong("RedSpawn"));

        ListTag list = tag.getList("CapturePoints", Tag.TAG_COMPOUND);
        for (int i = 0; i < list.size(); i++) {
            data.capturePoints.add(CapturePoint.load(list.getCompound(i)));
        }

        return data;
    }

    public static AASWorldData get(ServerLevel level) {
        return level.getDataStorage().computeIfAbsent(AASWorldData::load, AASWorldData::new, "aas_world_data");
    }

    public static class CapturePoint {
        public String name; public AABB area; public int bluePriority; public int redPriority;
        public int captureTimeMinutes; public int ticketPenalty;
        public String owner = "NEUTRAL"; public float progress = 0.0f;

        public CapturePoint(String name, AABB area, int bp, int rp, int time, int penalty) {
            this.name = name; this.area = area; this.bluePriority = bp; this.redPriority = rp;
            this.captureTimeMinutes = time; this.ticketPenalty = penalty;
        }

        public CompoundTag save() {
            CompoundTag tag = new CompoundTag();
            tag.putString("Name", name);
            tag.putDouble("MinX", area.minX); tag.putDouble("MinY", area.minY); tag.putDouble("MinZ", area.minZ);
            tag.putDouble("MaxX", area.maxX); tag.putDouble("MaxY", area.maxY); tag.putDouble("MaxZ", area.maxZ);
            tag.putInt("BlueP", bluePriority); tag.putInt("RedP", redPriority);
            tag.putInt("Time", captureTimeMinutes); tag.putInt("Penalty", ticketPenalty);
            tag.putString("Owner", owner);
            return tag;
        }

        public static CapturePoint load(CompoundTag tag) {
            AABB box = new AABB(tag.getDouble("MinX"), tag.getDouble("MinY"), tag.getDouble("MinZ"), tag.getDouble("MaxX"), tag.getDouble("MaxY"), tag.getDouble("MaxZ"));
            CapturePoint p = new CapturePoint(tag.getString("Name"), box, tag.getInt("BlueP"), tag.getInt("RedP"), tag.getInt("Time"), tag.getInt("Penalty"));
            p.owner = tag.getString("Owner");
            return p;
        }
    }
}
